cmake_minimum_required(VERSION 3.16)


# ============================================================================
# 1. COMPONENT CONFIGURATION
# ============================================================================

message(STATUS "[RideCast:APP] Configuring application component")


# ============================================================================
# 2. APPLICATION DIRECTORY CONFIGURATION
# ============================================================================

# Absolute path of the current ESP-IDF component.
set(RIDECAST_APP_MAIN_DIR "${CMAKE_CURRENT_LIST_DIR}")

# Parent directory containing all application modules.
get_filename_component(
    RIDECAST_APP_DIR
    "${RIDECAST_APP_MAIN_DIR}/.."
    ABSOLUTE
)

set(RIDECAST_APP_CONFIG_DIR
    "${RIDECAST_APP_DIR}/app_config"
)

set(RIDECAST_APP_INCLUDE_DIR
    "${RIDECAST_APP_DIR}/include"
)

set(RIDECAST_APP_INTERFACE_DIR
    "${RIDECAST_APP_DIR}/interface"
)

set(RIDECAST_APP_SRCS_DIR
    "${RIDECAST_APP_DIR}/srcs"
)

set(RIDECAST_APP_TEST_DIR
    "${RIDECAST_APP_DIR}/test"
)


# ============================================================================
# 3. APPLICATION CONFIGURATION SOURCES
# ============================================================================

set(RIDECAST_APP_CFG_SRCS
    "${RIDECAST_APP_CONFIG_DIR}/rc_config.c"
)


# ============================================================================
# 4. APPLICATION IMPLEMENTATION SOURCES
# ============================================================================
#
# These files are retained from the existing application configuration.
# Only files that exist will be registered.
#
# Add other application implementation files here as needed.
# ============================================================================

set(RIDECAST_APP_SRCS "")

set(RIDECAST_APP_CANDIDATE_SRCS
    "${RIDECAST_APP_SRCS_DIR}/uart_app.c"
    "${RIDECAST_APP_SRCS_DIR}/wifi_app.c"
    "${RIDECAST_APP_SRCS_DIR}/ridecast_main.c"
)

foreach(APP_SOURCE IN LISTS RIDECAST_APP_CANDIDATE_SRCS)

    if(EXISTS "${APP_SOURCE}")

        list(APPEND RIDECAST_APP_SRCS
            "${APP_SOURCE}"
        )

    else()

        message(STATUS
            "[RideCast:APP] Source not present: ${APP_SOURCE}"
        )

    endif()

endforeach()


# ============================================================================
# 5. APPLICATION INTERFACE SOURCES
# ============================================================================

set(RIDECAST_APP_INTERFACE_SRCS
    "${RIDECAST_APP_INTERFACE_DIR}/rc_uart.c"
)


# ============================================================================
# 6. APPLICATION ENTRY POINT
# ============================================================================

set(RIDECAST_APP_ENTRY_SRCS
    "${RIDECAST_APP_MAIN_DIR}/app_main.c"
)


# ============================================================================
# 7. APPLICATION TEST SOURCES
# ============================================================================
#
# Application tests are excluded from the firmware by default.
# They should be enabled explicitly when required.
# ============================================================================

set(RIDECAST_APP_TEST_SRCS "")

if(DEFINED RIDECAST_BUILD_APP_TESTS AND RIDECAST_BUILD_APP_TESTS)

    if(EXISTS "${RIDECAST_APP_TEST_DIR}/test_host_uart.c")

        list(APPEND RIDECAST_APP_TEST_SRCS
            "${RIDECAST_APP_TEST_DIR}/test_host_uart.c"
        )

    endif()

endif()


# ============================================================================
# 8. APPLICATION INCLUDE DIRECTORIES
# ============================================================================

set(RIDECAST_APP_INCLUDE_DIRS
    "${RIDECAST_APP_CONFIG_DIR}"
    "${RIDECAST_APP_INCLUDE_DIR}"
    "${RIDECAST_APP_INTERFACE_DIR}"
    "${CMAKE_SOURCE_DIR}/external/os/cmsis_esp/Include"
    "${CMAKE_BINARY_DIR}/version"
)


# ============================================================================
# 9. APPLICATION SOURCE VALIDATION
# ============================================================================

set(RIDECAST_REQUIRED_SRCS
    ${RIDECAST_APP_CFG_SRCS}
    ${RIDECAST_APP_INTERFACE_SRCS}
    ${RIDECAST_APP_ENTRY_SRCS}
)

foreach(APP_SOURCE IN LISTS RIDECAST_REQUIRED_SRCS)

    if(NOT EXISTS "${APP_SOURCE}")

        message(FATAL_ERROR
            "[RideCast:APP] Required source file not found: "
            "${APP_SOURCE}"
        )

    endif()

endforeach()


# ============================================================================
# 10. COMPLETE APPLICATION SOURCE LIST
# ============================================================================

set(RIDECAST_ALL_APP_SRCS
    ${RIDECAST_APP_CFG_SRCS}
    ${RIDECAST_APP_SRCS}
    ${RIDECAST_APP_INTERFACE_SRCS}
    ${RIDECAST_APP_ENTRY_SRCS}
    ${RIDECAST_APP_TEST_SRCS}
)


# ============================================================================
# 11. ESP-IDF COMPONENT REGISTRATION
# ============================================================================
#
# CLI and logger are provided by the BSP component.
#
# CMSIS-RTOS2 is provided by the cmsis_esp component.
# Do not compile cmsis_os2.c again in this component.
# ============================================================================

idf_component_register(

    SRCS
        ${RIDECAST_ALL_APP_SRCS}

    INCLUDE_DIRS
        ${RIDECAST_APP_INCLUDE_DIRS}

    PRIV_REQUIRES
        freertos
        cmsis_esp
        system
        lwip
        driver
        esp_netif
        esp_wifi
        esp_event
        log
        esp-tls
        esp_common
        bsp
        nvs_flash
        backoffAlgorithmWrapper
)


# ============================================================================
# 12. CONFIGURATION STATUS
# ============================================================================

message(STATUS "[RideCast:APP] ========================================")

message(STATUS "[RideCast:APP] Configuration source : rc_config.c")
message(STATUS "[RideCast:APP] Interface source     : rc_uart.c")
message(STATUS "[RideCast:APP] Application entry    : app_main.c")
message(STATUS "[RideCast:APP] BSP dependency       : Enabled")
message(STATUS "[RideCast:APP] CMSIS dependency     : cmsis_esp")

message(STATUS "[RideCast:APP] ========================================")
